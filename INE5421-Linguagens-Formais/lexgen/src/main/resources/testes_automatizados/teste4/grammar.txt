<Programa> ::= PROGRAM id ABRE_CHAVES <Declaracoes> <Comandos> FECHA_CHAVES
<Declaracoes> ::= <Decl> <Declaracoes>
<Declaracoes> ::= &
<Decl> ::= <Tipo> id PONTO_VIRGULA
<Tipo> ::= INT
<Tipo> ::= FLOAT
<Comandos> ::= <Cmd> <Comandos>
<Comandos> ::= &
<Cmd> ::= <Atribuicao>
<Cmd> ::= <Condicional>
<Cmd> ::= <Laço>
<Cmd> ::= <Leitura>
<Cmd> ::= <Escrita>
<Cmd> ::= <Bloco>
<Atribuicao> ::= id ATRIBUICAO <Expressao> PONTO_VIRGULA
<Condicional> ::= IF ABRE_PAR <Condicao> FECHA_PAR <Cmd> <SenaoOpcional>
<SenaoOpcional> ::= ELSE <Cmd>
<SenaoOpcional> ::= &
<Laço> ::= WHILE ABRE_PAR <Condicao> FECHA_PAR <Cmd>
<Leitura> ::= READ ABRE_PAR id FECHA_PAR PONTO_VIRGULA
<Escrita> ::= PRINT ABRE_PAR <Expressao> FECHA_PAR PONTO_VIRGULA
<Bloco> ::= ABRE_CHAVES <Comandos> FECHA_CHAVES
<Condicao> ::= <Expressao> <OpRel> <Expressao>
<OpRel> ::= IGUAL
<OpRel> ::= OP_MAIOR
<OpRel> ::= OP_MENOR
<Expressao> ::= <Termo> <RestoExpressao>
<RestoExpressao> ::= OP_MAIS <Termo> <RestoExpressao>
<RestoExpressao> ::= OP_MENOS <Termo> <RestoExpressao>
<RestoExpressao> ::= &
<Termo> ::= <Fator> <RestoTermo>
<RestoTermo> ::= OP_VEZES <Fator> <RestoTermo>
<RestoTermo> ::= OP_DIV <Fator> <RestoTermo>
<RestoTermo> ::= &
<Fator> ::= id
<Fator> ::= num_int
<Fator> ::= num_real
<Fator> ::= ABRE_PAR <Expressao> FECHA_PAR
