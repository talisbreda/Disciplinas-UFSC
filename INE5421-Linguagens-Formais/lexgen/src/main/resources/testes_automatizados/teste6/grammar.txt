<programa> ::= <declaracoes> <comandos> PONTO
<declaracoes> ::= <dcl_const> <dcl_var> <dcl_procs>
<dcl_const> ::= const <tipo_pre_def> id IGUAL <constante> PONTO_VIRGULA <dcl_const>
<dcl_const> ::= &
<dcl_var> ::= var <lid> DOIS_PONTOS <tipo> PONTO_VIRGULA <dcl_var>
<dcl_var> ::= &
<dcl_procs> ::= <dcl_proc> PONTO_VIRGULA <dcl_procs>
<dcl_procs> ::= &
<dcl_proc> ::= proc id <parametros> PONTO_VIRGULA <declaracoes> <comandos>
<dcl_proc> ::= funcao id <parametros> DOIS_PONTOS <tipo_pre_def> PONTO_VIRGULA <declaracoes> <comandos>
<parametros> ::= ABRE_PAR <listapar> FECHA_PAR
<parametros> ::= &
<listapar> ::= ref <lid> DOIS_PONTOS <tipo_pre_def> <rep_listapar>
<listapar> ::= val <lid> DOIS_PONTOS <tipo_pre_def> <rep_listapar>
<rep_listapar> ::= PONTO_VIRGULA <listapar>
<rep_listapar> ::= &
<lid> ::= id <rep_id>
<rep_id> ::= VIRGULA id <rep_id>
<rep_id> ::= &
<constante> ::= id
<constante> ::= <constante_explicita>
<tipo> ::= <tipo_pre_def>
<tipo> ::= intervalo <constante> PONTO_PONTO <constante>
<tipo> ::= cadeia ABRE_COL <constante> FECHA_COL
<tipo> ::= vetor ABRE_COL <constante> FECHA_COL de <tipo_pre_def>
<tipo_pre_def> ::= inteiro
<tipo_pre_def> ::= real
<tipo_pre_def> ::= booleano
<tipo_pre_def> ::= caracter
<comandos> ::= inicio <comando> <rep_comando> fim
<rep_comando> ::= PONTO_VIRGULA <comando> <rep_comando>
<rep_comando> ::= &
<comando> ::= id <rcomid>
<comando> ::= inicio <comando> <rep_comando> fim
<comando> ::= se <expressao> entao <comando> <senaoparte>
<comando> ::= enquanto <expressao> faca <comando>
<comando> ::= repita <comando> ate <expressao>
<comando> ::= leia ABRE_PAR <lid> FECHA_PAR
<comando> ::= escreva ABRE_PAR <expressao> <rep_expressao> FECHA_PAR
<comando> ::= &
<senaoparte> ::= senao <comando>
<senaoparte> ::= &
<rcomid> ::= ATRIBUICAO <expressao>
<rcomid> ::= ABRE_COL <expressao> FECHA_COL ATRIBUICAO <expressao>
<rcomid> ::= ABRE_PAR <expressao> <rep_expressao> FECHA_PAR
<rcomid> ::= &
<rep_expressao> ::= VIRGULA <expressao> <rep_expressao>
<rep_expressao> ::= &
<expressao> ::= <expsimp> <resto_expressao>
<resto_expressao> ::= <oprel> <expsimp>
<resto_expressao> ::= &
<oprel> ::= IGUAL
<oprel> ::= OP_MENOR
<oprel> ::= OP_MAIOR
<oprel> ::= OP_MAIOR_IGUAL
<oprel> ::= OP_MENOR_IGUAL
<oprel> ::= OP_DIFF
<expsimp> ::= <termo> <rep_expsimp>
<rep_expsimp> ::= <op_add> <termo> <rep_expsimp>
<rep_expsimp> ::= &
<op_add> ::= OP_MAIS
<op_add> ::= OP_MENOS
<op_add> ::= ou
<termo> ::= <fator> <rep_termo>
<rep_termo> ::= <op_mult> <fator> <rep_termo>
<rep_termo> ::= &
<op_mult> ::= OP_VEZES
<op_mult> ::= OP_DIV
<op_mult> ::= e
<fator> ::= nao <fator>
<fator> ::= OP_MENOS <fator>
<fator> ::= ABRE_PAR <expressao> FECHA_PAR
<fator> ::= <variavel>
<fator> ::= <constante_explicita>
<variavel> ::= id <rvar>
<rvar> ::= ABRE_PAR <expressao> <rep_expressao> FECHA_PAR
<rvar> ::= ABRE_COL <expressao> FECHA_COL
<rvar> ::= &
<constante_explicita> ::= num_int
<constante_explicita> ::= num_real
<constante_explicita> ::= falso
<constante_explicita> ::= verdadeiro
<constante_explicita> ::= literal
