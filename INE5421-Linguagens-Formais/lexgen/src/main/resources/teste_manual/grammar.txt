<S> ::= PROGRAM id PONTO_VIRGULA <Vars> <Bloco> PONTO
<Vars> ::= VAR <ListaDecl>
<Vars> ::= &
<ListaDecl> ::= <Decl> <ListaDecl>
<ListaDecl> ::= <Decl>
<Decl> ::= <ListaId> DOIS_PONTOS <Tipo> PONTO_VIRGULA
<ListaId> ::= id VIRGULA <ListaId>
<ListaId> ::= id
<Tipo> ::= INTEGER
<Tipo> ::= BOOLEAN
<Bloco> ::= BEGIN <Cmds> END
<Cmds> ::= <Cmd> PONTO_VIRGULA <Cmds>
<Cmds> ::= <Cmd>
<Cmds> ::= &
<Cmd> ::= id ASSIGN <Expr>
<Cmd> ::= IF <Expr> THEN <Cmd> <ElsePart>
<Cmd> ::= FOR id ASSIGN <Expr> TO <Expr> DO <Cmd>
<Cmd> ::= WRITELN ABRE_PAR <Expr> FECHA_PAR
<Cmd> ::= <Bloco>
<ElsePart> ::= ELSE <Cmd>
<ElsePart> ::= &
<Expr> ::= <Termo> <RestoExpr>
<RestoExpr> ::= MAIS <Termo> <RestoExpr>
<RestoExpr> ::= MENOS <Termo> <RestoExpr>
<RestoExpr> ::= MAIOR <Termo> <RestoExpr>
<RestoExpr> ::= MENOR <Termo> <RestoExpr>
<RestoExpr> ::= IGUAL <Termo> <RestoExpr>
<RestoExpr> ::= &
<Termo> ::= <Fator> <RestoTermo>
<RestoTermo> ::= VEZES <Fator> <RestoTermo>
<RestoTermo> ::= DIV <Fator> <RestoTermo>
<RestoTermo> ::= &
<Fator> ::= id
<Fator> ::= num
<Fator> ::= TRUE
<Fator> ::= FALSE
<Fator> ::= ABRE_PAR <Expr> FECHA_PAR
